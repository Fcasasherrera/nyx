import tools , ConexionBD

def lambda_handler(event, context):
    #Validaciones
    keys= [
        { 'nombre': 'accion', 'tipo': str},
        { 'nombre': 'cliente', 'tipo': str}
    ]

    try:
        #Se obtienen argumentos desde body
        args = tools.obtenerArgumentos(event)
        #Se valida que no este vacio args por falta de parametros o no era json
        if len(args) == 0: 
            return tools.responseJson(400,"No se recibieron parámetros")
        tools.validarParametros(args,keys)
        if len(args['cliente']) < 9 or len(args['accion']) == 0:
            return tools.responseJson(400,"Información no válida")
            
        #Se establece conexion con base de datos
        conexionIntelisis= ConexionBD.Conexion("intelisistmp","Mavicubos")
        
        sql = f"""
                SET NOCOUNT ON;
                EXEC SPERM0906_UpDateDocsCobroCteFinal 
                ?, ?
                """
        parms =  (args['cliente'],args['accion'])  
        
        result = conexionIntelisis.procedureNoResult(sql,parms)
        if result['statusCode'] != 200:
           return tools.responseJson(result['statusCode'],result["result"])

        return tools.responseJson(result['statusCode'],[{"resp":"Proceso ralizado"}])
    except Exception as e:
        return tools.responseJson(402,e,Default=str)
